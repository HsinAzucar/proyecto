== Disparadores entregas

|------
|Nombre|Tiempo|Evento|Definición
|------
|actualizar_calificacion_entrega|AFTER|INSERT|BEGIN
    DECLARE promedio DECIMAL(5,2);

    -- Calcular promedio de todas las entregas del alumno en esa materia
    SELECT AVG(e.calificacion)
    INTO promedio
    FROM entregas e
    INNER JOIN tareas t ON e.id_tarea = t.id_tarea
    WHERE e.id_alumno = NEW.id_alumno
      AND t.id_materia = (SELECT id_materia FROM tareas WHERE id_tarea = NEW.id_tarea);

    -- Si ya existe calificación para ese alumno y materia, actualizar
    IF EXISTS (
        SELECT 1 FROM calificaciones 
        WHERE id_alumno = NEW.id_alumno 
          AND id_materia = (SELECT id_materia FROM tareas WHERE id_tarea = NEW.id_tarea)
    ) THEN
        UPDATE calificaciones
        SET calificacion = promedio
        WHERE id_alumno = NEW.id_alumno
          AND id_materia = (SELECT id_materia FROM tareas WHERE id_tarea = NEW.id_tarea);
    ELSE
        -- Si no existe, insertar nueva calificación
        INSERT INTO calificaciones (id_alumno, id_materia, calificacion, periodo)
        VALUES (
            NEW.id_alumno,
            (SELECT id_materia FROM tareas WHERE id_tarea = NEW.id_tarea),
            promedio,
            &#039;2026-A&#039; -- Ajusta el periodo según tu sistema
        );
    END IF;
END
